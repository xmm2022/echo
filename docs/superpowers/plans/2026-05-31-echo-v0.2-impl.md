# Echo v0.2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Echo v0.2 as an Emby front proxy that rewrites mapped PlaybackInfo sources to `/emby/stream/{token}` or `/emby/error/{token}`, with DB users, user-aware account pools, quota accounting, and corrected sidecar copy-failure handling.

**Architecture:** Keep v0.1 resource-pool modules intact and add v0.2 application-layer packages around them. `/api/*` remains Echo management/v0.1 compatibility; Emby clients only receive `/emby/stream/{token}` or `/emby/error/{token}` for controlled playback. Each phase lands as one commit and must pass a read-only gate review before the next phase.

**Tech Stack:** Go 1.22, chi, sqlc, SQLite migrations under `internal/store/schema`, templ+htmx for the existing UI, slog redaction, Prometheus metrics.

## Non-Negotiable Invariants

- PlaybackInfo rewrite must never emit `/api/restore/{file_id}` or `/api/stream/{file_id}`.
- `/emby/stream/{token}` and `/emby/error/{token}` must be Echo-reserved routes and must never fall through to upstream Emby.
- Ordinary `/emby/*` proxy must not become a playback bypass. Suspicious playback/download/transcode endpoints fail closed until Phase 4 mapping guard tests prove safe behavior.
- Mapped source playable URL fields are either Echo stream URLs or Echo error URLs. They never remain upstream Emby URLs.
- Mapping cache is not authorization cache. PlaybackInfo rewrite and `/emby/stream` both re-check current owner/grant authorization.
- Echo does not save or refresh raw 115/139/189 cookies.
- Do not import sidecar Go packages. Keep HTTP REST process boundary.

```go
type SidecarErrorKind string

const (
	SidecarErrProtocol         SidecarErrorKind = "protocol"
	SidecarErrTransport        SidecarErrorKind = "transport"
	SidecarErrAPIEnvelope      SidecarErrorKind = "api_envelope"
	SidecarErrObjectMissing    SidecarErrorKind = "object_missing"
	SidecarErrStorageMissing   SidecarErrorKind = "storage_missing"
	SidecarErrStorageUnhealthy SidecarErrorKind = "storage_unhealthy"
	SidecarErrAuthOrAccount    SidecarErrorKind = "auth_or_account"
	SidecarErrTransient        SidecarErrorKind = "transient"
)

type SidecarTypedError struct {
	Kind         SidecarErrorKind
	Operation    string
	HTTPStatus   int
	OpenListCode int
	SafeMessage  string
	RawMessage   string
	EvidenceClass string // json_envelope, http_status, html_snippet, transport
	Confidence    string // confirmed, suspect, low
}

func (e *SidecarTypedError) Error() string {
	if e == nil {
		return "<nil>"
	}
	if e.SafeMessage != "" {
		return fmt.Sprintf("sidecar %s failed: %s", e.Operation, e.SafeMessage)
	}
	return fmt.Sprintf("sidecar %s failed: %s", e.Operation, e.Kind)
}
```

```go
func classifySidecarMessage(operation string, httpStatus, openListCode int, msg string) *SidecarTypedError {
	safe := safeSidecarMessage(msg)
	safe = strings.TrimSpace(safe)
	if len(safe) > 512 {
		safe = safe[:512]
	}
	lower := strings.ToLower(safe)
	kind := SidecarErrTransient
	confidence := "low"
	evidence := "json_envelope"
	switch {
	case strings.Contains(lower, "object not found") || strings.Contains(lower, "failed to get file"):
		kind = SidecarErrObjectMissing
		if operation == "link" && httpStatus == http.StatusOK && openListCode != 200 && openListCode != 0 {
			confidence = "confirmed"
		} else {
			confidence = "suspect"
		}
	case strings.Contains(lower, "storage not found") || strings.Contains(lower, "please add a storage first"):
		kind = SidecarErrStorageMissing
		confidence = "suspect"
	case strings.Contains(lower, "access_token is empty") || strings.Contains(lower, "refresh token failed") || strings.Contains(lower, "unauthorized") || strings.Contains(lower, "forbidden"):
		kind = SidecarErrAuthOrAccount
		confidence = "suspect"
	case httpStatus == http.StatusUnauthorized || httpStatus == http.StatusForbidden:
		kind = SidecarErrAuthOrAccount
		evidence = "http_status"
		confidence = "suspect"
	case openListCode == 0:
		kind = SidecarErrProtocol
		confidence = "low"
	}
	return &SidecarTypedError{
		Kind:         kind,
		Operation:    operation,
		HTTPStatus:   httpStatus,
		OpenListCode: openListCode,
		SafeMessage:  safe,
		RawMessage:   msg,
		EvidenceClass: evidence,
		Confidence:    confidence,
	}
}
```

```go
package auth

type UserContext struct {
	UserID string
	Role   string
	Scopes []string
	Now    time.Time

func GenerateToken() (string, error) {
	selector, err := randomBase64URL(12)
	if err != nil {
		return "", err
	}
	secret, err := randomBase64URL(32)
	if err != nil {
		return "", err
	}
	return selector + "." + secret, nil
}

func HashToken(token string) string {
	sum := sha256.Sum256([]byte(token))
	return "sha256:" + hex.EncodeToString(sum[:])
}

func VerifyTokenHash(token, stored string) bool {
	got := HashToken(token)
	return subtle.ConstantTimeCompare([]byte(got), []byte(stored)) == 1
}

func (u UserContext) HasScope(scope string) bool {
	if u.Role == "admin" && contains(u.Scopes, "admin") {
		return scope == "admin" || scope == "read" || scope == "playback"
	}
	return contains(u.Scopes, scope)
}
```

```go
var (
	ErrUnauthorized = errors.New("playback unauthorized")
	ErrEntryMissing = errors.New("library entry missing")

type Resolver struct {
	q   Querier
	now func() time.Time
	streamTimeout time.Duration
}

type Querier interface {
	GetLibraryEntryByID(ctx context.Context, arg queries.GetLibraryEntryByIDParams) (queries.LibraryEntry, error)
	UserCanPlaybackLibrary(ctx context.Context, arg queries.UserCanPlaybackLibraryParams) (bool, error)
	ListPlayableCopiesForUser(ctx context.Context, arg queries.ListPlayableCopiesForUserParams) ([]queries.ListPlayableCopiesForUserRow, error)
}

func NewResolver(q Querier, now func() time.Time) *Resolver

func (r *Resolver) ResolveCopies(ctx context.Context, echoUserID string, libraryEntryID int64, preferProvider string, limit int64) ([]queries.ListPlayableCopiesForUserRow, error) {
	entry, err := r.q.GetLibraryEntryByID(ctx, queries.GetLibraryEntryByIDParams{ID: libraryEntryID})
	if errors.Is(err, sql.ErrNoRows) {
		return nil, ErrEntryMissing
	}
	if err != nil {
		return nil, err
	}
	allowed, err := r.q.UserCanPlaybackLibrary(ctx, queries.UserCanPlaybackLibraryParams{LibraryID: entry.LibraryID, EchoUserID: echoUserID})
	if err != nil {
		return nil, err
	}
	if !allowed {
		return nil, ErrUnauthorized
	}
	if limit <= 0 || limit > 5 {
		limit = 5
	}
rows, err := r.q.ListPlayableCopiesForUser(ctx, queries.ListPlayableCopiesForUserParams{
		BlobID: entry.BlobID, EchoUserID: echoUserID, PreferProvider: preferProvider,
		Now: r.now().Unix(), ActiveSince: r.now().Add(-r.streamTimeout).Unix(),
		DayStart: dayStart(r.now()).Unix(), DayEnd: dayStart(r.now()).Add(24 * time.Hour).Unix(),
		Limit: limit,
	})
	if err != nil {
		return nil, err
	}
	return stableWeightedOrder(rows, echoUserID, libraryEntryID, "")
}
```

```go
const PathNormVersion = 1

type MappingRule struct {
	ID            int64
	LibraryID     int64
	PrefixNorm    string
	EchoRelPrefix string
	CaseSensitive bool
}

type MappingResult struct {
	MappingID int64
	LibraryID int64
	RelPath   string
	PathNorm  string
}

func NormalizeEmbyPath(raw string) (string, error)
func MatchPath(rules []MappingRule, rawPath string) (MappingResult, error)
```

Rules:
- Convert `\` to `/` before validation.
- Percent-decode exactly once using `url.PathUnescape`.
- Reject raw or decoded slash/backslash escapes (`%2f`, `%5c`), NUL/control characters, `..` segments, empty path, Windows drive prefixes, UNC paths, and Unicode slash-like runes.
- Preserve ordinary spaces and non-slash Unicode filename characters.
- Longest normalized prefix wins, but prefix matching must respect path segment boundaries: `/media` matches `/media/Film.mkv`, not `/media2/Film.mkv`.
- For case-insensitive rules, compare lowercase normalized path/prefix but keep output rel path casing from the input.
- Join `echo_rel_prefix` with relative suffix using path-style separators only; reject an `echo_rel_prefix` that is absolute, contains `..`, or introduces an empty/UNC-like path.

```go
	if body != "" {

	}
	req := httptest.NewRequest(method, target, bodyReader)
	if body != "" {
		req.Header.Set("Content-Type", "application/json")

	req = req.WithContext(auth.NewContext(req.Context(), user))
	rec := httptest.NewRecorder()
	r.ServeHTTP(rec, req)
```

```go
func requireAdmin(w http.ResponseWriter, r *http.Request) (auth.UserContext, bool) {
	user := auth.FromContext(r.Context())
	if !user.HasScope("admin") {
		writeAPIError(w, http.StatusForbidden, "forbidden")
		return auth.UserContext{}, false
	}
	return user, true
}

func requestedUserOrSelf(w http.ResponseWriter, r *http.Request) (string, bool) {
	user := auth.FromContext(r.Context())
	requested := strings.TrimSpace(r.URL.Query().Get("user_id"))
	if requested == "" {
		return user.UserID, true
	}
	if requested != user.UserID && !user.HasScope("admin") {
		writeAPIError(w, http.StatusForbidden, "forbidden")
		return "", false
	}
	return requested, true
}
```

Handler rules:
- Admin-only endpoints call `requireAdmin`.
- Owner-scoped endpoints use `requestedUserOrSelf`; `/api/quota/usage` must compute usage from Phase 2 authoritative `playback_events` for the selected policy period with `GetQuotaUsageFromEvents`, not from the future/cache `quota_usage` table.
