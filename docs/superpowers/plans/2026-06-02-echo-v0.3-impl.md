# Echo v0.3 Discovery Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Echo v0.3 discovery: TMDB subscriptions, Telegram/poster source crawling, rule scoring, admin review, and 115-only dispatch into the existing `ingest_producer` pipeline.

**Architecture:** Discovery is an internal Echo application layer. Source adapters emit normalized candidates, rules score candidates per subscription, dispatch enqueues existing `ingest_producer` jobs, and v0.1 ingest remains the only writer of `library_entries` and `file_copies`.

**Tech Stack:** Go 1.22, SQLite, sqlc, chi, templ, htmx, Prometheus metrics, `gotd/td` isolated under `internal/discovery/sources/telegram`, existing `115share2cas` producer.

```sql
  subscription_id   INTEGER REFERENCES discovery_subscriptions(id),

  actor_user_id  TEXT NOT NULL REFERENCES users(id),
  request_id     TEXT,

CREATE INDEX idx_discovery_sources_due ON discovery_sources(enabled, scheduler_state, next_run_at, locked_until);
```

```sql
  SELECT id FROM discovery_sources
  WHERE enabled = 1
    AND scheduler_state IN ('healthy','backoff')
```

```go
package secrets

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

func Resolve(root, ref string) (string, error) {
	rel, ok := strings.CutPrefix(ref, "ref:")
	if !ok || rel == "" {
		return "", fmt.Errorf("secret ref must use ref:relative/path")
	}
	if filepath.IsAbs(rel) {
		return "", fmt.Errorf("secret ref must be relative")
	}
	clean := filepath.Clean(rel)
	if clean == "." || clean == ".." || strings.HasPrefix(clean, ".."+string(os.PathSeparator)) {
		return "", fmt.Errorf("secret ref must not escape root")
	}
	base, err := filepath.Abs(root)
	if err != nil {
		return "", err
	}
	if resolvedBase, err := filepath.EvalSymlinks(base); err == nil {
		base = resolvedBase
	}
	target := filepath.Join(base, clean)
	resolvedTarget, err := filepath.EvalSymlinks(target)
	if err != nil {
		return "", err
	}
	prefix := base + string(os.PathSeparator)
	if resolvedTarget != base && !strings.HasPrefix(resolvedTarget, prefix) {
		return "", fmt.Errorf("secret ref resolves outside root")
	}
	info, err := os.Stat(resolvedTarget)
	if err != nil {
		return "", err
	}
	if !info.Mode().IsRegular() {
		return "", fmt.Errorf("secret ref must resolve to a regular file")
	}
	return resolvedTarget, nil
}

func ResolveEnv(ref string) (string, error) {
	name, ok := strings.CutPrefix(ref, "env:")
	if !ok || name == "" {
		return "", fmt.Errorf("env ref must use env:NAME")
	}
	value, ok := os.LookupEnv(name)
	if !ok {
		return "", fmt.Errorf("environment variable %s is not set", name)
	}
	return value, nil
}

func ValidateRef(root, ref string) error {
	if strings.HasPrefix(ref, "env:") {
		name := strings.TrimPrefix(ref, "env:")
		if name == "" {
			return fmt.Errorf("env ref must use env:NAME")
		}
		return nil
	}
	rel, ok := strings.CutPrefix(ref, "ref:")
	if !ok || rel == "" {
		return fmt.Errorf("secret ref must use env:NAME or ref:relative/path")
	}
	if filepath.IsAbs(rel) {
		return fmt.Errorf("secret ref must be relative")
	}
	clean := filepath.Clean(rel)
	if clean == "." || clean == ".." || strings.HasPrefix(clean, ".."+string(os.PathSeparator)) {
		return fmt.Errorf("secret ref must not escape root")
	}
	return nil
}

func RedactRef(ref string) string {
	if ref == "" {
		return ""
	}
	return "[REDACTED]"
}
```

```go
func (g RealDiscoveryGate) Run(ctx context.Context) (RealDiscoveryResult, error) {
	st, err := store.Open(filepath.Join(os.TempDir(), "echo-real-discovery-"+strconv.FormatInt(time.Now().UnixNano(), 10)+".db"))
	if err != nil {
		return RealDiscoveryResult{}, err
	}
	defer st.Close()
	ds := discovery.NewStore(st)
	now := time.Now().Unix()
	if _, err := st.DB.ExecContext(ctx, `
INSERT OR IGNORE INTO accounts (
  id, provider, sidecar_id, storage_mount, status, owner_id, created_at, updated_at
) VALUES (?, '115', ?, ?, 'active', 'admin', ?, ?)`,
		g.AccountID, g.SidecarID, g.StorageMount, now, now); err != nil {
		return RealDiscoveryResult{}, err
	}
	var libraryID int64
	if err := st.DB.QueryRowContext(ctx, `
INSERT INTO libraries (name, echo_output_kind, echo_output_path, owner_id, created_at)
VALUES ('real discovery gate', 'local', ?, 'admin', ?)
RETURNING id`, g.LibraryOutputPath, now).Scan(&libraryID); err != nil {
		return RealDiscoveryResult{}, err
	}
	var producerProfileID int64
	if err := st.DB.QueryRowContext(ctx, `
INSERT INTO discovery_producer_profiles (
  name, provider, tool, target_account, target_subdir_template,
  library_rel_path_template, default_args_json, enabled, created_at, updated_at
) VALUES (
  'real 115', '115', '115share2cas', ?, '{{.Title}}', '{{.Title}}',
  json_object('cookie_file', ?, 'recycle_password_file', ?), 1, ?, ?
) RETURNING id`, g.AccountID, g.CookieRef, g.RecycleRef, now, now).Scan(&producerProfileID); err != nil {
		return RealDiscoveryResult{}, err
	}
	var ruleProfileID int64
	if err := st.DB.QueryRowContext(ctx, `
INSERT INTO rule_profiles (name, version, rules_json, enabled, created_at, updated_at)
VALUES ('real gate default', 1, '{"weights":["resolutions","colors","audios","extensions"]}', 1, ?, ?)
RETURNING id`, now, now).Scan(&ruleProfileID); err != nil {
		return RealDiscoveryResult{}, err
	}
	sourceID, subscriptionID, err := seedRealDiscoveryRows(ctx, st, RealSeedRows{
		SessionRef: g.SessionRef, Channel: g.Channel, TMDBID: g.TMDBID,
		LibraryID: libraryID, ProducerProfileID: producerProfileID,
		RuleProfileID: ruleProfileID, HistoryLimit: g.HistoryLimit, Now: now,
	})
	if err != nil {
		return RealDiscoveryResult{}, err
	}
	tmdbKey, err := resolveGateSecret(g.SecretsRoot, g.TMDBAPIKeyRef)
	if err != nil {
		return RealDiscoveryResult{}, err
	}
	tgAPIHash, err := resolveGateSecret(g.SecretsRoot, g.TGAPIHashRef)
	if err != nil {
		return RealDiscoveryResult{}, err
	}
	orch := discovery.NewOrchestrator(discovery.Deps{
		Store: ds,
		SourceAdapters: map[discovery.SourceKind]discovery.SourceAdapter{
			discovery.SourceTelegramMTProto: telegram.NewAdapter(
				telegram.NewMTProtoClient(g.SecretsRoot, g.TGAPIID, tgAPIHash),
			),
		},
		Enqueue: realGateEnqueuer(ctx, st, g),
		ProducerConfig: realGateProducerConfig(g),
	})
	if err := orch.RunSourceCrawl(ctx, discovery.SourceCrawlPayload{SourceID: sourceID}); err != nil {
		return RealDiscoveryResult{}, err
	}
	if _, err := tmdb.NewClient(tmdb.Config{APIKey: tmdbKey, Language: "zh-CN"}, tmdb.NewSQLiteCache(st.Queries, "zh-CN")).MovieDetails(ctx, g.TMDBID); err != nil {
		return RealDiscoveryResult{}, err
	}
	if err := orch.RunSubscriptionCheck(ctx, discovery.SubscriptionCheckPayload{SubscriptionID: subscriptionID}); err != nil {
		return RealDiscoveryResult{}, err
	}
	matchID, err := acceptBestRealGateMatch(ctx, ds, subscriptionID)
	if err != nil {
		return RealDiscoveryResult{}, err
	}
	if err := orch.RunDispatch(ctx, discovery.DispatchPayload{MatchID: matchID}); err != nil {
		return RealDiscoveryResult{}, err
	}
	if err := runQueuedProducerJob(ctx, st, g); err != nil {
		return RealDiscoveryResult{}, err
	}
	if err := orch.RunReconcile(ctx, discovery.ReconcilePayload{MatchID: matchID, JobID: latestProducerJobID(ctx, st)}); err != nil {
		return RealDiscoveryResult{}, err
	}
	return RealDiscoveryResult{
		Candidates: countRealGateCandidates(ctx, st),
		ProducerJobs: countRealGateProducerJobs(ctx, st),
		EchoFilesWritten: countEchoFiles(g.LibraryOutputPath),
	}, nil
}
```

`seedRealDiscoveryRows` inserts one `discovery_sources` row with `kind='telegram_mtproto'`, `secret_ref=rows.SessionRef`, and `config_json` containing `{"channel": rows.Channel, "history_limit": rows.HistoryLimit}`; one `telegram_channels` row for the channel; one `tmdb_media` row or refresh marker for `rows.TMDBID`; and one active `discovery_subscriptions` row using the provided library, producer profile, and rule profile ids.

`realGateProducerConfig` returns an `ingest.ProducerConfig` whose `Tools["115share2cas"]` binary is `g.ProducerBinary` and whose allowlist includes `share_url`, `share_code`, `receive_code`, `cookie_file`, `recycle_password_file`, and `mode`. `runQueuedProducerJob` executes the real `job.KindIngestProducer` handler against the configured sidecar, so the gate proves discovery dispatch still enters v0.1 ingest rather than writing library rows directly.
